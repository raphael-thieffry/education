Rattraper une exception :
try
	expressioninitiale
with
	| Exception1 -> expr1
	| Exception2 (s) -> expr2
-> on fait l'expression initiale
-> si pas d'erreur, on renvoie l'expression
-> si une erreur, si elle est rattrapée, on fait l'expression qui correspond
->		  si elle n'est pas rattrapée, elle se propage

Opérations enregistrements (struct) :

définition :
	type nomEnregistrement = {
		nomAtt1 : typeAtt1 ;
		nomAtt2 : typeAtt2
	}
-> définition du nom et du type des attributs
-> le point virgule est optionnel à la fin de la dernière ligne

création d'une instance :
	let nomInstance = {
		nomAtt1 = valeurAtt1 ;
		nomAtt2 = valeurAtt2
	}
-> comme une affectation des attributs

accès aux champs :
	nomInstance.nomAtt

=> ATTENTION : ne pas donner les mêmes noms de champs (si argument de fonction, comment on peut deviner le champ de quel enregistrement est demandé ?)


Opérations type somme (enum) :

définition :
	type nomSomme = nomTypePerso1 | nomTypePerso2 | nomTypePerso3 | etc.
-> l'un des types peut être une valeur de type nomTypePersoN mais de valeur un type (int par ex) :
	type nomSomme = etc | nomTypeN of int | NomTypeNN of int*int
-> nomTypePerso1, 2, etc. = CONSTANTES du type somme
-> nomTypePersoN = VALEUR ETIQUETÉE

création d'une instance :
	let att_a = nomType1
	let att_b = nomTypeN (3)
	let att_c = nomTypeNN(3,5)
-> On peut mettre n'importe quel élément du type choisi
-> Attention, att_b n'est pas de type int mais de type nomSomme

manipulation de types sommes :
-> motif (pattern) :
	nomTypeCst
	nomValeurEtiquetee (3) : avec ou sans les parenthèses
	nomValeurEtiquetee (n) : n est la variable qui contient la valeur pour l'expression qui suit et n est de type donné (int par ex)
	_ : n'importe quelle valeur du type somme

Syntaxe :
	match attDeTypeNomSomme with
	motif1 -> expr1
	| motif2 -> expr2
	| motif3 | motif4 -> exprFactorisee
	| motifEtiquetteN (3) -> expression si l'entier stocké vaut 3
	| motifEtiqueteN (n) -> expression si l'entier est autre, et sa valeur est stockée dans n pour la suite
	| motifEtiqueteNN (m, _) -> expression où osef de 2e elem
	| _ -> exprDefaut
-> nom1 -> expr1 : appelé branche
-> c'est un "sinon si" donc ne pas se tromper d'ordre (3 puis n et pas n puis 3)
-> on peut factoriser les cas

-> on peut faire un match d'un n-uplet
	match att1, att2 with
	nom1_CsteY, nom2_CsteX -> ceci
	nom1_CsteA, nom2_CsteA | nom2_Cste2, nom2_CsteB -> cela
	nom1_Cste, _ -> cela
	_, nom2_Etiquettee val -> ceci avec val dedans
